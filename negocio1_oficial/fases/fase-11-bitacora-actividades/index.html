Bitácora de Auditoría con Fecha y Hora

Registro persistente de operaciones clave con actor, acción y fecha local 'es-BO' en localStorage.

📍 Aplicación: Se aplica en el panel administrativo del gerente para rastrear cambios de precios y pedidos.

Historial de Auditoría:

Código Implementado en esta Fase: 
let logs = JSON.parse(localStorage.getItem('fase_logs') || '[]');
        function renderLogs() {
            const c = document.getElementById('listaBitacora');
            c.innerHTML = logs.map(l => '<div style="padding:6px; border-bottom:1px solid var(--line); font-size:12px;"><b>' + l.actor + '</b>: ' + l.accion + ' <small style="color:var(--muted);">(' + l.fecha + ')</small></div>').join('') || '<p>Sin eventos.</p>';
        }
        function addLog(accion) {
            logs.unshift({ actor: 'Gerente Admin', accion, fecha: new Date().toLocaleString('es-BO') });
            localStorage.setItem('fase_logs', JSON.stringify(logs.slice(0, 10)));
            renderLogs();
        }
        document.getElementById('btnLogPedido').addEventListener('click', () => addLog('Aprobó comanda #104'));
        document.getElementById('btnLogPrecio').addEventListener('click', () => addLog('Modificó precio de almuerzo a 20 Bs'));
        renderLogs();
