Filtros y Búsqueda en Tiempo Real

Filtrado reactivo en tiempo real por coincidencia de texto e inclusión de categorías sin recargar la página.

📍 Aplicación: Se aplica en el buscador de platos del menú semanal y filtros de especiales.

Código Implementado en esta Fase: 
const lista = [
            "Lunes: Sopa de arroz con maní", "Lunes: Milanesa de pollo", "Lunes: Saice",
            "Martes: Sopa de fideo", "Martes: Ají de lenteja", "Martes: Pollo al horno",
            "Miércoles: Chairo", "Miércoles: Chuleta de chancho",
            "Jueves: Sopa de maní", "Jueves: Trucha", "Jueves: Falso conejo"
        ];
        const input = document.getElementById('buscarPlato');
        const res = document.getElementById('resultadosBusqueda');
        function render(items) {
            res.innerHTML = items.length ? items.map(t => '<div style="padding:10px; background:var(--card); border:1px solid var(--line); border-radius:6px;">' + t + '</div>').join('') : '<p>No se encontraron platos.</p>';
        }
        render(lista);
        input.addEventListener('input', () => {
            const query = input.value.toLowerCase().trim();
            render(lista.filter(item => item.toLowerCase().includes(query)));
        });
